function handleKeyPress(e) { if (document.getElementById('passcodeModal').classList.contains('active')) return; if (e.key === ' ') e.preventDefault(); let keyId = e.key === ' ' ? 'key-space' : `key-${e.key.toLowerCase()}`; let pressedKeyEl = document.getElementById(keyId); if (pressedKeyEl) pressedKeyEl.classList.add('pressed'); if (e.key === 'Enter' && document.getElementById('completionModal').classList.contains('active')) { handleModalAction(); return; } // GAME LESSON HANDLER if (lessons[currentLessonIdx].type === 'game') { const accuracy = RocketGame.handleInput(e.key); if (accuracy !== undefined) updateAccuracyMeter(accuracy); return; } if (lessons[currentLessonIdx].type === 'video' || e.key.length !== 1 || e.ctrlKey || e.altKey || e.metaKey) return; // TYPING LESSON HANDLER const expectedChar = targetText[currentIndex]; const isUppercaseExpected = (expectedChar === expectedChar.toUpperCase() && expectedChar !== ' ' && expectedChar !== ';'); let isValid = false; // Check if we are currently on Lesson 6 (Index 5) const isLesson6 = (currentLessonIdx === 5); if (isUppercaseExpected && isLesson6) { // STRICT SHIFT RULE (Only enforced on Lesson 6) // Right-hand keys (J, K) require Left Shift // Left-hand keys (F, D) require Right Shift const isRightHandLetter = ['j', 'k', 'l', ';', 'u', 'i', 'o', 'p', 'h', 'n', 'm'].includes(expectedChar.toLowerCase()); const usedCorrectShift = isRightHandLetter ? leftShiftDown : rightShiftDown; isValid = (e.key === expectedChar && usedCorrectShift); } else if (isUppercaseExpected) { // General uppercase check for any other lesson (accepts either Shift key) isValid = (e.shiftKey && e.key === expectedChar); } else { isValid = (e.key === expectedChar); } totalTyped++; if (isValid) { correctTyped++; currentIndex++; } else { errorIndices.push(currentIndex); if (currentLessonIdx < 3) currentIndex++; } const acc = Math.round((correctTyped / totalTyped) * 100); updateAccuracyMeter(acc); updateTargetAndKeyboard(); if (currentIndex >= targetText.length) showCompletionModal(acc); }
