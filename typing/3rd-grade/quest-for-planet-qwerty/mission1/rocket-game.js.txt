// rocket-game.js
const RocketGame = (() => {
  let active = false;
  let targetLetter = 'F';
  let destroyed = 0;
  const totalGoal = 20;
  let totalTyped = 0;
  let correctTyped = 0;
  let onCompleteCallback = null;

  const letters = ['F', 'J', 'D', 'K'];

  function init(onComplete) {
    active = true;
    destroyed = 0;
    totalTyped = 0;
    correctTyped = 0;
    onCompleteCallback = onComplete;
    updateUI();
    spawnNext();
  }

  function reset() {
    active = false;
    destroyed = 0;
  }

  function spawnNext() {
    targetLetter = letters[Math.floor(Math.random() * letters.length)];
    const rocketLetterEl = document.getElementById('rocketLetter');
    const targetRocketEl = document.getElementById('targetRocket');
    if (rocketLetterEl) rocketLetterEl.innerText = targetLetter;

    if (targetRocketEl) {
      const randomX = Math.floor(Math.random() * 70) + 15;
      targetRocketEl.style.left = `${randomX}%`;
    }
  }

  function handleInput(key) {
    if (!active) return undefined;

    totalTyped++;
    const pressedChar = key.toUpperCase();

    if (pressedChar === targetLetter) {
      correctTyped++;
      destroyed++;
      updateUI();

      if (destroyed >= totalGoal) {
        active = false;
        const finalAccuracy = Math.round((correctTyped / totalTyped) * 100);
        if (onCompleteCallback) onCompleteCallback(finalAccuracy);
      } else {
        spawnNext();
      }
    } else {
      const targetRocketEl = document.getElementById('targetRocket');
      if (targetRocketEl) {
        targetRocketEl.classList.add('shake');
        setTimeout(() => targetRocketEl.classList.remove('shake'), 200);
      }
    }

    return Math.round((correctTyped / totalTyped) * 100);
  }

  function updateUI() {
    const launchedEl = document.getElementById('launchedCount');
    const targetEl = document.getElementById('targetGoal');
    if (launchedEl) launchedEl.innerText = destroyed;
    if (targetEl) targetEl.innerText = totalGoal;
  }

  return { init, reset, handleInput };
})();
